Task 2: Tool Callbacks Implementation (`student_challenge/callbacks/callback_tools.py`)

Implement security governance and latency tracking inside before_tool_callback and after_tool_callback:

  1. Record Start Time: Store time.perf_counter() in tool_context.state[f"tool_start_time:{tool_name}"].
  2. RBAC Authorization Check: Read user_id = tool_context.state.get("user_id"). If user_id is unauthorized (e.g. "guest"), return a block dictionary: {"status": "BLOCKED", "error": f"User '{user_id}' is not authorized."}.
  3. Business Guardrail: Inspect args (such as exchange_code) and ensure the exchange is in approved list ["CME", "CBOT", "NYMEX", "COMEX"]. Block invalid inputs.
  4. Latency Calculation: In after_tool_callback, calculate elapsed time duration_ms = (time.perf_counter() - start_time) * 1000 and record in tool_context.state["last_tool_duration_ms"].

Complete `callbacks/callback_tools.py`

Python: student_challenge/callbacks/callback_tools.py 
import time
import logging
from typing import Any, Dict, Optional
from google.adk import Context

logger = logging.getLogger("exchange_agent")

AUTHORIZED_USERS = {"alice": True, "bob": True, "guest": False}
ALLOWED_EXCHANGES = {"CME", "CBOT", "NYMEX", "COMEX"}

def before_tool_callback(
    tool: Any,
    args: Dict[str, Any],
    tool_context: Context,
) -> Optional[Dict[str, Any]]:

    tool_name = getattr(tool, "name", str(tool))

    # 1. Record start time for latency tracking
    tool_context.state[f"tool_start_time:{tool_name}"] = time.perf_counter()

    # 2. Authorization Check (RBAC)
    user_id = tool_context.state.get("user_id", "guest")
    if not AUTHORIZED_USERS.get(user_id, False):
        return {"status": "BLOCKED", "error": f"User '{user_id}' is not authorized to call tools."}

    # 3. Exchange Whitelist Guardrail
    exchange = (args or {}).get("exchange_code", "").strip().upper()
    if exchange and exchange not in ALLOWED_EXCHANGES:
        return {"status": "BLOCKED", "error": f"Exchange '{exchange}' is not in allowed list: {sorted(ALLOWED_EXCHANGES)}"}

    print(f"[BEFORE TOOL] Intercepting: {tool_name} with args {args}")
    return None


def after_tool_callback(
    tool: Any,
    args: Dict[str, Any],
    tool_context: Context,
    tool_response: Any,
) -> Optional[Dict[str, Any]]:

    tool_name = getattr(tool, "name", str(tool))
    start_time = tool_context.state.get(f"tool_start_time:{tool_name}")

    if start_time:
        duration_ms = (time.perf_counter() - start_time) * 1000
        tool_context.state["last_tool_duration_ms"] = duration_ms
        logger.info(f"Tool '{tool_name}' completed in {duration_ms:.2f} ms")

    print(f"[AFTER TOOL] {tool_name} completed successfully")
    return None

Updated `student_challenge/agent.py`

Python: student_challenge/agent.py 
from google.adk import Agent, Workflow, Context
from .tools.custom_tools import get_exchange_details, list_products_by_exchange
from .callbacks.callback_tools import before_tool_callback, after_tool_callback
from .callbacks.callback_agents import before_agent_callback, after_agent_callback

MODEL = "gemini-2.5-flash"

def save_original_request(ctx: Context, node_input: str) -> str:
    ctx.state["original_request"] = node_input
    ctx.state["user_id"] = "alice"  # Set to 'alice' for authorized access
    return node_input

exchange_info_agent = Agent(
    name="exchange_info_agent",
    model=MODEL,
    mode="single_turn",
    description="Specialist for exchange details (name, location, description).",
    tools=[get_exchange_details],
    before_agent_callback=before_agent_callback,
    after_agent_callback=after_agent_callback,
    before_tool_callback=before_tool_callback,
    after_tool_callback=after_tool_callback,
)

product_info_agent = Agent(
    name="product_info_agent",
    model=MODEL,
    mode="single_turn",
    description="Specialist for products belonging to CME exchanges.",
    tools=[list_products_by_exchange],
    before_agent_callback=before_agent_callback,
    after_agent_callback=after_agent_callback,
    before_tool_callback=before_tool_callback,
    after_tool_callback=after_tool_callback,
)

supervisor_agent = Agent(
    name="exchange_support_supervisor",
    model=MODEL,
    description="Supervisor coordinating exchange and product specialists.",
    sub_agents=[exchange_info_agent, product_info_agent],
)

root_agent = Workflow(
    name="cme_exchange_support",
    edges=[("START", save_original_request, supervisor_agent)],
)
